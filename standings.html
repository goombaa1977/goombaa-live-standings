function getCurrentDisplayData() { let targetData = []; // 2027 and future/blank periods stay empty if (currentYear === '2027') { targetData = []; } else if (currentTab === 'yearly') { if (currentMonth === 'ALL') { targetData = allData['yearly'] || []; } else { // This maps August (and other months) to look for your backend's specific keys, // falling back to the active monthly dataset if it matches current active periods. const monthKey = `yearly_${currentYear}_${currentMonth}`; const shortMonthKey = currentMonth.toLowerCase(); if (allData[monthKey]) { targetData = allData[monthKey]; } else if (allData[shortMonthKey]) { targetData = allData[shortMonthKey]; } else if (currentMonth === 'AUG' && allData['monthly']) { // If August is the current active month, use the monthly standings data targetData = allData['monthly']; } else { targetData = []; } } } else { targetData = allData[currentTab] || []; } renderRows(targetData); } function renderRows(playerList) { const container = document.getElementById('standings-rows'); if (!playerList || playerList.length === 0) { container.innerHTML = `
No standings recorded for this period.
`; return; } const sortedList = [...playerList].sort((a, b) => { const winsA = parseInt(a.wins, 10) || 0; const winsB = parseInt(b.wins, 10) || 0; return winsB - winsA; }); let htmlContent = ''; sortedList.forEach(p => { const winsCount = p.wins !== undefined ? p.wins : '0'; const tier = getTierInfo(winsCount); htmlContent += `
${p.tag || p.gamertag}
${p.platform || 'Twitch'}
${winsCount}
${tier.name}
`; }); container.innerHTML = htmlContent; }
